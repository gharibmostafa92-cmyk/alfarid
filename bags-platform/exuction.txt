function InvApp() {
  const [t, setTweak] = useTweaks(window.INV_TWEAK_DEFAULTS);
  const [nav, setNav] = useState(() => {
    try { return JSON.parse(localStorage.getItem('vortix-inv-route')) || { route: 'stock', params: {} }; } catch { return { route: 'stock', params: {} }; }
  });
  const [db, setDb] = useState(() => ({ branches: INV_BRANCHES, locations: INV_LOCATIONS, transfers: INV_TRANSFERS, counts: INV_COUNTS, production: INV_PRODUCTION }));
  const [toastMsg, setToastMsg] = useState(null);
  const [reloadKey, setReloadKey] = useState(0);
  const tRef = useRef();

  const go = (route, params = {}) => { setNav({ route, params }); window.scrollTo({ top: 0, behavior: 'smooth' }); };
  useEffect(() => { localStorage.setItem('vortix-inv-route', JSON.stringify(nav)); }, [nav]);
  const toast = text => { setToastMsg({ text }); clearTimeout(tRef.current); tRef.current = setTimeout(() => setToastMsg(null), 2400); };

  const ctx = { db, setDb, toast, role: t.role, demoState: t.demoState, reloadKey, reload: () => { setReloadKey(k => k + 1); if (t.demoState === 'error') setTweak('demoState', 'live'); } };

  const screens = {
    branches: BranchesScreen, locations: LocationsScreen, stock: StockScreen, adjust: AdjustScreen, moves: MovesScreen,
    transfers: TransfersScreen, transfer: TransferDetail, 'transfer-edit': TransferEdit, counts: CountsScreen, count: CountDetail, production: ProductionScreen,
  };
  const Screen = screens[nav.route] || StockScreen;
  const screenKey = nav.route + JSON.stringify(nav.params);
  const label = (INV_NAV.find(n => n.match.includes(nav.route)) || {}).label;

  return (
    <InvCtx.Provider value={ctx}>
      <div id="app-viewport" className={t.theme === 'dark' ? 'theme-dark' : 'theme-light'}>
        <div className="max-w-[1440px] mx-auto px-4 sm:px-7 pt-5 pb-10">
          <InvHeader route={nav.route} go={go} theme={t.theme} role={t.role} />
          <div className="flex gap-6 items-start">
            <InvSidebar route={nav.route} go={go} theme={t.theme} setTheme={v => setTweak('theme', v)} />
            <main className="flex-1 min-w-0" data-screen-label={label}>
              <Screen key={screenKey} go={go} params={nav.params} />
            </main>
          </div>
        </div>
        <Toast toast={toastMsg} />
      </div>

      <div dir="ltr">
      <TweaksPanel title="Tweaks">
        <TweakSection label="الصلاحية">
          <TweakRadio label="الدور" value={t.role} onChange={v => setTweak('role', v)}
            options={[{ value: 'admin', label: 'مدير' }, { value: 'storekeeper', label: 'أمين مخزن' }, { value: 'viewer', label: 'عرض فقط' }]} />
        </TweakSection>
        <TweakSection label="حالة البيانات">
          <TweakSelect label="محاكاة الاستجابة" value={t.demoState} onChange={v => setTweak('demoState', v)}
            options={[{ value: 'live', label: 'بيانات فعلية' }, { value: 'loading', label: 'تحميل' }, { value: 'empty', label: 'فارغ' }, { value: 'error', label: 'خطأ من السيرفر' }]} />
        </TweakSection>
        <TweakSection label="المظهر">
          <TweakRadio label="الوضع" value={t.theme} onChange={v => setTweak('theme', v)} options={[{ value: 'light', label: 'فاتح' }, { value: 'dark', label: 'داكن' }]} />
        </TweakSection>
      </TweaksPanel>
      </div>
    </InvCtx.Provider>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<InvApp />);

const INV_USERS = ['أحمد سمير', 'منى خالد', 'كريم عادل', 'سارة فؤاد'];

function invGenMoves() {
  const out = [];
  const types = Object.keys(INV_MOVE_TYPES);
  const refs = { adj_in: 'ADJ', adj_out: 'ADJ', dmg_mark: 'ADJ', dmg_writeoff: 'ADJ', dmg_restore: 'ADJ', tr_out: 'TRF', tr_in: 'TRF', count: 'CNT', production: 'PRD', sale: 'INV' };
  let seed = 7;
  const rnd = () => { seed = (seed * 9301 + 49297) % 233280; return seed / 233280; };
  for (let i = 0; i < 64; i++) {
    const st = INV_STOCK[Math.floor(rnd() * INV_STOCK.length)];
    const t = types[Math.floor(rnd() * types.length)];
    const mt = INV_MOVE_TYPES[t];
    const qty = st.variantId === 'v9' ? +(rnd() * 40 + 1).toFixed(2) : Math.ceil(rnd() * 24);
    const sign = mt.sign === 0 ? (rnd() > .5 ? 1 : -1) : mt.sign;
    const before = Math.round(st.onHand + rnd() * 30);
    const day = 23 - Math.floor(i / 5);
    const h = String(8 + (i * 7) % 12).padStart(2, '0'), m = String((i * 13) % 60).padStart(2, '0');
    out.push({
      id: 'm' + i, date: `2026-10-${String(day).padStart(2, '0')} ${h}:${m}`,
      ref: `${refs[t]}-2026-${String(1480 - i * 3).padStart(5, '0')}`,
      variantId: st.variantId, branchId: st.branchId, locId: st.locId, type: t,
      qty: sign * qty, before, after: +(before + sign * qty).toFixed(4),
      user: INV_USERS[i % 4], unitCost: invVariant(st.variantId).cost,
      note: ['', 'كرتونة مفتوحة', 'مطابقة رصيد افتتاحي', '', 'تلف أثناء النقل', '', 'طلب فرع'][i % 7],
    });
  }
  return out;
}

    const e = {};
    if (!f.name.trim()) e.name = 'الاسم مطلوب';
    if (!f.code.trim()) e.code = 'الكود مطلوب';

  const [type, setType] = useState('all');

  const [del, setDel] = useState(null);
  const [delBusy, setDelBusy] = useState(false);
  const [delErr, setDelErr] = useState(null);

  const locCount = id => db.locations.filter(l => l.branchId === id).length;
  const rows = db.branches.filter(b => (type === 'all' || b.type === type) && (!q || b.name.includes(q) || b.code.toLowerCase().includes(q.toLowerCase()) || b.city.includes(q)));

  const save = f => {
    if (f.id) setDb(d => ({ ...d, branches: d.branches.map(b => b.id === f.id ? f : b) }));
    else setDb(d => ({ ...d, branches: [{ ...f, id: 'b' + Date.now() }, ...d.branches] }));

    toast(f.id ? 'تم حفظ التعديلات' : 'تمت إضافة ' + (f.type === 'warehouse' ? 'المخزن' : 'الفرع'));

    setTimeout(() => {
      setDelBusy(false);
      const lc = locCount(del.id), sc = INV_STOCK.filter(s => s.branchId === del.id).length;

        setDelErr(`لا يمكن حذف «${del.name}» لارتباطه بـ ${lc} موقع تخزين${sc ? ` و ${sc} رصيد مخزون` : ''}. يمكنك تعطيله بدلًا من الحذف. (409 Conflict)`);

                <button key={k} className={type === k ? 'on' : ''} onClick={() => setType(k)}>{l} <span className="num c-muted mr-1">{counts[k]}</span></button>
              ))}

        <Phase phase={phase} cols={7}

function CountCreate({ open, onClose, go }) {
  const { setDb, toast } = useInv();
  const [branchId, setBranchId] = useState(null);
  const [locId, setLocId] = useState(null);
  const [note, setNote] = useState('');
  const [mode, setMode] = useState('all');
  const [picked, setPicked] = useState([]);
  const [tried, setTried] = useState(false);
  useEffect(() => { if (open) { setBranchId(null); setLocId(null); setNote(''); setMode('all'); setPicked([]); setTried(false); } }, [open]);

  const scope = branchId ? INV_STOCK.filter(s => s.branchId === branchId && (!locId || s.locId === locId)) : [];
  const items = mode === 'all' ? [...new Set(scope.map(s => s.variantId))] : picked;
  const valid = branchId && items.length > 0 && items.length <= CNT_MAX;

  const submit = () => {
    setTried(true); if (!valid) return;
    const id = 'c' + Date.now();
    setDb(d => ({ ...d, counts: [{ id, no: 'CNT-2026-000' + (32 + d.counts.length - 3), branchId, locId, status: 'counting', createdBy: 'أحمد سمير', date: '2026-10-23', note,
      lines: items.map(v => ({ variantId: v, system: scope.filter(s => s.variantId === v).reduce((a, s) => a + s.onHand, 0), actual: null })) }, ...d.counts] }));
    toast('تم إنشاء الجرد'); onClose(); go('count', { id });
  };

  return (
    <Modal open={open} onClose={onClose} width="max-w-2xl" title="جرد جديد" sub={`حتى ${CNT_MAX} بند عند الإنشاء`}
      footer={<><button className="btn btn-dark" onClick={submit}>إنشاء وبدء العدّ</button><button className="btn btn-soft" onClick={onClose}>إلغاء</button></>}>
      <div className="space-y-4">
        <div className="grid grid-cols-2 gap-4">
          <Field label="الفرع" req error={tried && !branchId ? 'مطلوب' : null}><Sel value={branchId} onChange={v => { setBranchId(v); setLocId(null); setPicked([]); }} placeholder="اختر…" options={branchOpts(b => b.active)} /></Field>
          <Field label="الموقع" hint="اختياري"><Sel value={locId} onChange={setLocId} placeholder={branchId ? 'كل المواقع' : '—'} disabled={!branchId} options={branchId ? locOpts(branchId) : []} /></Field>
        </div>
        <div>
          <span className="field-label">الأصناف</span>
          <div className="seg mb-3">
            <button className={mode === 'all' ? 'on' : ''} onClick={() => setMode('all')}>كل الأصناف في النطاق</button>
            <button className={mode === 'pick' ? 'on' : ''} onClick={() => setMode('pick')}>اختيار أصناف محددة</button>
          </div>
          {mode === 'all' ? (
            <div className="v-subcard p-3.5 flex items-center justify-between">
              <span className="text-[12.5px] c-secondary">{branchId ? 'سيتم إدراج كل SKU له رصيد في النطاق المحدد' : 'اختر الفرع لعرض عدد الأصناف'}</span>
              <span className="num font-display text-[20px] font-bold c-primary">{items.length}</span>
            </div>
          ) : (
            <div className="space-y-2">
              <VariantPicker value={null} onChange={v => v && setPicked(p => p.length < CNT_MAX ? [...p, v] : p)} exclude={picked} placeholder="أضف SKU…" />
              <div className="flex flex-wrap gap-1.5">
                {picked.map(v => <span key={v} className="badge b-gray !py-1"><span className="mono">{invVariant(v).sku}</span><button onClick={() => setPicked(p => p.filter(x => x !== v))}><Icon n="x" s={11} /></button></span>)}
                {picked.length === 0 && <span className="text-[11.5px] c-muted">لم تتم إضافة أصناف</span>}
              </div>
            </div>
          )}
          {tried && branchId && items.length === 0 && <div className="text-[11px] mt-1.5" style={{ color: 'var(--badge-red-text)' }}>لا توجد أصناف للجرد</div>}
          <div className="text-[11px] c-muted mt-1.5 num">{items.length} / {CNT_MAX}</div>
        </div>
        <Field label="ملاحظات"><textarea className="inp" rows={2} value={note} onChange={e => setNote(e.target.value)} /></Field>
      </div>
    </Modal>
  );
}

function CountDetail({ go, params }) {
  const { db, setDb, toast, demoState } = useInv();
  const can = useCan();
  const phase = useLoad('cnt' + params.id);
  const c = db.counts.find(x => x.id === params.id);
  const [vals, setVals] = useState(() => Object.fromEntries((c?.lines || []).map((l, i) => [i, l.actual === null ? '' : String(l.actual)])));
  const [filter, setFilter] = useState('all');
  const [confirm, setConfirm] = useState(false);
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState(null);
  const [dirty, setDirty] = useState(false);
  if (!c) return null;

  const editable = c.status !== 'approved' && can('count.manage');
  const actual = i => vals[i] === '' || vals[i] === undefined ? null : parseFloat(vals[i]);
  const rows = c.lines.map((l, i) => ({ ...l, i, a: actual(i), diff: actual(i) === null ? null : +(actual(i) - l.system).toFixed(4) }));
  const counted = rows.filter(r => r.a !== null).length;
  const shown = rows.filter(r => filter === 'all' || (filter === 'uncounted' && r.a === null) || (filter === 'diff' && r.diff) || (filter === 'match' && r.diff === 0));
  const diffs = rows.filter(r => r.diff);

  const saveCounts = () => { setDb(d => ({ ...d, counts: d.counts.map(x => x.id === c.id ? { ...x, status: 'counting', lines: x.lines.map((l, i) => ({ ...l, actual: actual(i) })) } : x) })); setDirty(false); toast('تم حفظ العدّ'); };
  const approve = () => {
    setBusy(true); setErr(null);
    setTimeout(() => {
      setBusy(false);
      if (demoState === 'error') { setErr('تم تنفيذ حركات على بعض الأصناف بعد إنشاء الجرد؛ أعد احتساب كميات النظام قبل الاعتماد. (409)'); return; }
      setDb(d => ({ ...d, counts: d.counts.map(x => x.id === c.id ? { ...x, status: 'approved', lines: x.lines.map((l, i) => ({ ...l, actual: actual(i) })) } : x) }));
      setConfirm(false); toast('تم اعتماد الجرد وترحيل الفروق');
    }, 900);
  };

  return (
    <section className="space-y-5">
      <PageHead title={<span className="mono">{c.no}</span>} back={{ label: 'الجرد', onClick: () => go('counts') }}
        sub={<span className="flex items-center gap-2 flex-wrap">{invBranch(c.branchId).name} <span className="c-muted">·</span> {c.locId ? <span className="mono">{invLoc(c.locId).code}</span> : 'كل المواقع'} <Badge cls={INV_CNT_STATUS[c.status].cls} dot>{INV_CNT_STATUS[c.status].label}</Badge></span>}
        actions={c.status !== 'approved' && <>
          {editable && <button className="btn btn-soft" disabled={!dirty} onClick={saveCounts}>حفظ العدّ</button>}
          {can('count.approve')
            ? <button className="btn btn-orange" onClick={() => { setErr(null); setConfirm(true); }} disabled={counted < rows.length}><Icon n="check" s={14} w={2.2} /> اعتماد الجرد</button>
            : <Can perm="count.approve" mode="الاعتماد يتطلب صلاحية" />}
        </>} />

      <Phase phase={phase} cols={5} rows={5} empty={<Card><EmptyState icon="clipboard" title="لا توجد بنود في هذا الجرد" /></Card>}>
        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3.5 stagger-item stagger-2">
          <KPI hot label="تم العدّ" icon="clipboard" value={`${counted}/${rows.length}`} sub={counted < rows.length ? `${rows.length - counted} بند لم يُعدّ بعد` : 'اكتمل العدّ'} />
          <KPI label="مطابق" icon="check" value={rows.filter(r => r.diff === 0).length} />
          <KPI label="بفروق" icon="alert" value={diffs.length} />
          <KPI label="صافي الفرق" icon="sliders" value={(() => { const n = diffs.reduce((a, r) => a + r.diff, 0); return (n > 0 ? '+' : '') + invFmt(n); })()} sub="بالوحدات" />
        </div>

        <Card stagger={3}>
          <CardHead title="جدول العدّ" right={
            <div className="seg">
              {[['all', 'الكل'], ['uncounted', 'لم يُعدّ'], ['diff', 'بفروق'], ['match', 'مطابق']].map(([k, l]) => <button key={k} className={filter === k ? 'on' : ''} onClick={() => setFilter(k)}>{l}</button>)}
            </div>} />
          {counted < rows.length && c.status !== 'approved' && <div className="mb-3"><InlineAlert kind="amber" icon="info" title="الحقل الفارغ ≠ صفر">الخانة الفارغة المظللة تعني «لم يتم العدّ». إذا كان الصنف غير موجود فعليًا اكتب 0 أو اضغط «صفر».</InlineAlert></div>}
          {shown.length === 0 ? <EmptyState icon="filter" title="لا توجد بنود بهذا الفلتر" /> : (
            <table className="vt">
              <thead><tr><th>SKU</th><th>الصنف</th><th className="t-num">كمية النظام</th><th className="t-num" style={{ width: 200 }}>الكمية الفعلية</th><th className="t-num">الفرق</th><th>الحالة</th></tr></thead>
              <tbody>
                {shown.map(r => (
                  <tr key={r.i}>
                    <td><Sku id={r.variantId} /></td>
                    <td><VariantCell id={r.variantId} showBarcode /></td>
                    <td className="t-num num c-secondary">{invFmt(r.system)}</td>
                    <td className="t-num">
                      {editable ? (
                        <div className="flex items-center gap-1.5 justify-end">
                          {r.a === null && <button className="btn btn-ghost btn-sm !px-2 !py-1 text-[11px]" onClick={() => { setVals(v => ({ ...v, [r.i]: '0' })); setDirty(true); }}>صفر</button>}
                          <div className={`rounded-[10px] w-28 ${r.a === null ? 'uncounted' : ''}`}>
                            <NumInput small value={vals[r.i]} placeholder="لم يُعدّ" onChange={v => { setVals(x => ({ ...x, [r.i]: v })); setDirty(true); }} className={r.a === null ? '!bg-transparent' : ''} />
                          </div>
                          {r.a !== null && <button className="icon-btn !w-6 !h-6" title="مسح (لم يُعدّ)" onClick={() => { setVals(v => ({ ...v, [r.i]: '' })); setDirty(true); }}><Icon n="x" s={11} /></button>}
                        </div>
                      ) : <span className="num font-semibold c-primary">{r.a === null ? <span className="c-muted font-normal">لم يُعدّ</span> : invFmt(r.a)}</span>}
                    </td>
                    <td className="t-num num font-bold" style={{ color: r.diff === null ? 'var(--text-muted)' : r.diff > 0 ? 'var(--badge-green-text)' : r.diff < 0 ? 'var(--badge-red-text)' : 'var(--text-secondary)' }}>
                      {r.diff === null ? '—' : r.diff === 0 ? '0' : (r.diff > 0 ? '+' : '−') + invFmt(Math.abs(r.diff))}
                    </td>
                    <td>{r.a === null ? <Badge cls="b-gray">لم يُعدّ</Badge> : r.a === 0 ? <Badge cls={r.diff ? 'b-red' : 'b-green'}>صفر فعلي</Badge> : r.diff === 0 ? <Badge cls="b-green">مطابق</Badge> : <Badge cls={r.diff > 0 ? 'b-blue' : 'b-red'}>{r.diff > 0 ? 'زيادة' : 'عجز'}</Badge>}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          )}
        </Card>
      </Phase>

      <ConfirmDialog open={confirm} onClose={() => setConfirm(false)} onConfirm={approve} busy={busy} error={err} title="اعتماد الجرد" confirmLabel="اعتماد وترحيل الفروق"
        body={<>سيتم ترحيل <b className="num c-primary">{diffs.length}</b> فرق كحركات «فرق جرد» وتحديث الأرصدة. لا يمكن تعديل العدّ بعد الاعتماد.</>} />
    </section>
  );
}

function ProductionForm({ open, onClose, onSave }) {
  const { demoState } = useInv();
  const [f, setF] = useState({ batch: '', branchId: null, locId: null, note: '' });
  const [lines, setLines] = useState([newPrLine()]);
  const [tried, setTried] = useState(false);
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState(null);
  useEffect(() => { if (open) { setF({ batch: '', branchId: null, locId: null, note: '' }); setLines([newPrLine()]); setTried(false); setErr(null); } }, [open]);
  const upd = (k, p) => setLines(ls => ls.map(l => l.key === k ? { ...l, ...p } : l));
  const bad = l => !l.variantId || !(parseInt(l.qty) > 0) || !(parseFloat(l.cost) >= 0 && l.cost !== '');
  const valid = f.batch.trim() && f.branchId && lines.every(l => !bad(l));
  const total = lines.reduce((a, l) => a + (parseInt(l.qty) || 0) * (parseFloat(l.cost) || 0), 0);
  const submit = () => {
    setTried(true); if (!valid) return;
    setBusy(true); setErr(null);
    setTimeout(() => {
      setBusy(false);
      if (demoState === 'error') { setErr('رقم الدفعة B-2610-07 مسجّل مسبقًا. (422 Unprocessable)'); return; }
      onSave({ id: 'pr' + Date.now(), no: 'PRD-2026-00089', ...f, date: '2026-10-23', user: 'أحمد سمير', lines: lines.map(l => ({ variantId: l.variantId, qty: parseInt(l.qty), cost: parseFloat(l.cost) })) });
    }, 800);
  };
  return (
    <Modal open={open} onClose={onClose} width="max-w-4xl" title="تسجيل استلام إنتاج" sub={`حتى ${PRD_MAX} بندًا · الكميات أعداد صحيحة`}

          <Field label="رقم الدفعة" req error={tried && !f.batch.trim() ? 'مطلوب' : null}><input className={`inp mono text-left ${tried && !f.batch.trim() ? 'err' : ''}`} value={f.batch} onChange={e => setF({ ...f, batch: e.target.value.toUpperCase() })} placeholder="B-2610-08" /></Field>
          <Field label="الفرع / المخزن" req error={tried && !f.branchId ? 'مطلوب' : null}><Sel value={f.branchId} onChange={v => setF({ ...f, branchId: v, locId: null })} placeholder="اختر…" options={branchOpts(b => b.active)} /></Field>
          <Field label="الموقع" hint="اختياري"><Sel value={f.locId} onChange={v => setF({ ...f, locId: v })} placeholder={f.branchId ? 'الموقع الافتراضي' : '—'} disabled={!f.branchId} options={f.branchId ? locOpts(f.branchId) : []} /></Field>

            <span className="field-label !mb-0">البنود <span className="num c-muted">{lines.length}/{PRD_MAX}</span></span>
            <button className="btn btn-soft btn-sm" disabled={lines.length >= PRD_MAX} onClick={() => setLines(ls => [...ls, newPrLine()])}><Icon n="plus" s={13} /> إضافة بند</button>

              {lines.map((l, i) => (
                <tr key={l.key}>
                  <td className="num c-muted">{i + 1}</td>
                  <td><div className={tried && !l.variantId ? 'rounded-[12px] ring-1 ring-[var(--badge-red-text)]' : ''}><VariantPicker small value={l.variantId} onChange={v => upd(l.key, { variantId: v, cost: l.cost || (v ? String(invVariant(v).cost) : '') })} /></div></td>
                  <td><NumInput small integer value={l.qty} onChange={v => upd(l.key, { qty: v })} err={tried && !(parseInt(l.qty) > 0)} /></td>

function TransferEdit({ go, params }) {
  const { db, setDb, toast } = useInv();
  const can = useCan();
  const existing = params.id ? db.transfers.find(t => t.id === params.id) : null;
  const [from, setFrom] = useState(existing?.from || null);
  const [to, setTo] = useState(existing?.to || null);
  const [note, setNote] = useState(existing?.note || '');
  const [lines, setLines] = useState(() => existing ? existing.lines.map(l => newTrLine({ ...l, req: String(l.req) })) : [newTrLine()]);
  const [tried, setTried] = useState(false);
  const [busy, setBusy] = useState(false);

  if (!can('transfers.create')) return <NoPerm title="تحويل جديد" perm="إنشاء التحويلات" go={go} />;

  const same = from && to && from === to;
  const upd = (key, p) => setLines(ls => ls.map(l => l.key === key ? { ...l, ...p } : l));
  const lineBad = l => !l.variantId || !(parseFloat(l.req) > 0);
  const valid = from && to && !same && lines.length && lines.every(l => !lineBad(l));

  const save = () => {
    setTried(true); if (!valid) return;
    setBusy(true);
    setTimeout(() => {
      const data = { from, to, note, lines: lines.map(l => ({ variantId: l.variantId, fromLoc: l.fromLoc, toLoc: l.toLoc, req: parseFloat(l.req), appr: null, ship: null, recv: null })) };
      let id = existing?.id;
      if (existing) setDb(d => ({ ...d, transfers: d.transfers.map(t => t.id === id ? { ...t, ...data } : t) }));
      else { id = 't' + Date.now(); setDb(d => ({ ...d, transfers: [{ id, no: 'TRF-2026-00' + (143 + d.transfers.length - 5), status: 'draft', date: '2026-10-23', createdBy: 'أحمد سمير', ...data }, ...d.transfers] })); }
      toast('تم حفظ التحويل كمسودة'); go('transfer', { id });
    }, 600);
  };

  const avail = (vid, br, loc) => INV_STOCK.filter(s => s.variantId === vid && s.branchId === br && (!loc || s.locId === loc)).reduce((a, s) => a + s.available, 0);

  return (
    <section className="space-y-5">
      <PageHead title={existing ? `تعديل ${existing.no}` : 'تحويل جديد'} back={{ label: 'التحويلات المخزنية', onClick: () => go(existing ? 'transfer' : 'transfers', existing ? { id: existing.id } : {}) }}
        actions={<><button className="btn btn-soft" onClick={() => go('transfers')}>إلغاء</button><button className="btn btn-dark" onClick={save} disabled={busy}>{busy && <span className="w-3.5 h-3.5 rounded-full border-2 border-current border-t-transparent animate-spin" />} حفظ كمسودة</button></>} />

      <Card stagger={2}>
        <div className="grid grid-cols-1 md:grid-cols-[1fr_auto_1fr] gap-4 items-end">
          <Field label="فرع المصدر" req error={tried && !from ? 'مطلوب' : null}>
            <Sel value={from} onChange={v => { setFrom(v); setLines(ls => ls.map(l => ({ ...l, fromLoc: null }))); }} placeholder="من…" options={branchOpts(b => b.active)} className={(tried && !from) || same ? 'err' : ''} />
          </Field>
          <button type="button" onClick={() => { setFrom(to); setTo(from); setLines(ls => ls.map(l => ({ ...l, fromLoc: l.toLoc, toLoc: l.fromLoc }))); }} className="w-10 h-10 rounded-full v-subcard flex items-center justify-center c-orange mx-auto mb-0.5" title="تبديل">

          </button>

          </Field>

        <Field label="ملاحظات" className="mt-4"><textarea className="inp" rows={2} value={note} onChange={e => setNote(e.target.value)} placeholder="سبب التحويل أو تعليمات للشحن…" /></Field>

    setTimeout(() => { setBusy(false); setDb(d => ({ ...d, transfers: d.transfers.map(x => x.id === t.id ? { ...x, status: 'cancelled' } : x) })); setConfirm(null); toast('تم إلغاء التحويل'); }, 600);

              <div className="overflow-x-auto">
                <table className="vt">
                  <thead><tr><th>المتغير</th><th>من ← إلى</th><th className="t-num">المطلوبة</th><th className="t-num">المعتمدة</th><th className="t-num">المشحونة</th><th className="t-num">المستلمة</th></tr></thead>
                  <tbody>
                    {t.lines.map((l, i) => {
                      const cell = (col) => {
                        if (stage && stageMeta[stage].col === col) {
                          const max = parseFloat(l[stageMeta[stage].srcCol]);
                          const over = parseFloat(qtys[i]) > max;
                          return <td className="t-num" style={{ width: 110 }}><NumInput small value={qtys[i]} onChange={v => setQtys(q => ({ ...q, [i]: v }))} err={over} className="!bg-[var(--accent-orange-light)]" />{over && <div className="text-[10px] mt-0.5" style={{ color: 'var(--badge-red-text)' }}>الحد {invFmt(max)}</div>}</td>;
                        }
                        const v = l[col]; const diff = col === 'recv' && v !== null && l.ship !== null && v !== l.ship;
                        return <td className="t-num num" style={{ color: v === null ? 'var(--text-muted)' : diff ? 'var(--badge-red-text)' : 'var(--text-primary)', fontWeight: v === null ? 400 : 600 }}>{v === null ? '—' : invFmt(v)}{diff && <span className="text-[10px] block font-normal">عجز {invFmt(l.ship - v)}</span>}</td>;
                      };
                      return (
                        <tr key={i}>
                          <td><div className="flex items-center gap-2.5"><Sku id={l.variantId} /><VariantCell id={l.variantId} /></div></td>
                          <td className="mono text-[11px] c-secondary">{l.fromLoc ? invLoc(l.fromLoc).code : '—'} → {l.toLoc ? invLoc(l.toLoc).code : '—'}</td>
                          <td className="t-num num font-semibold c-primary">{invFmt(l.req)}</td>
                          {cell('appr')}{cell('ship')}{cell('recv')}
                        </tr>
                      );
                    })}
                  </tbody>
                </table>
              </div>
              {stage && (
                <div className="flex items-center gap-2.5 mt-4 pt-4" style={{ borderTop: '1px solid var(--divider-color)' }}>
                  <button className="btn btn-orange" onClick={() => {
                    const m = stageMeta[stage];
                    if (t.lines.some((l, i) => parseFloat(qtys[i]) > parseFloat(l[m.srcCol]))) return;
                    setErr(null); setConfirm('stage');
                  }}><Icon n="check" s={14} w={2.2} /> تأكيد ال{stageMeta[stage].label}</button>
                  <button className="btn btn-soft" onClick={() => setStage(null)}>تراجع</button>
                </div>
              )}
            </Card>
          </div>

          <Card className="lg:col-span-4" stagger={4}>
            <h3 className="font-display text-[15px] font-semibold c-primary mb-3">التفاصيل</h3>
            <div className="space-y-3 text-[12.5px]">
              {[['المصدر', invBranch(t.from).name], ['الوجهة', invBranch(t.to).name], ['التاريخ', <span className="num">{t.date}</span>], ['أنشأه', t.createdBy]].map(([k, v]) => (
                <div key={k} className="flex justify-between"><span className="c-secondary">{k}</span><span className="font-medium c-primary">{v}</span></div>
              ))}
            </div>
            <div className="v-subcard p-3.5 mt-4 text-[12px] c-secondary">{t.note || <span className="c-muted">لا توجد ملاحظات</span>}</div>
            <div className="mt-4 space-y-2">
              {[['req', 'المطلوب'], ['appr', 'المعتمد'], ['ship', 'المشحون'], ['recv', 'المستلم']].map(([k, l]) => {
                const total = t.lines.reduce((a, x) => a + (x[k] || 0), 0); const req = t.lines.reduce((a, x) => a + x.req, 0);
                return (
                  <div key={k}>
                    <div className="flex justify-between text-[11.5px] mb-1"><span className="c-secondary">{l}</span><span className="num font-semibold c-primary">{invFmt(total)}</span></div>
                    <div className="h-1.5 rounded-full hatched overflow-hidden"><div className="h-full rounded-full bar-hatched-orange" style={{ width: (req ? total / req * 100 : 0) + '%' }} /></div>
                  </div>
                );
              })}
            </div>
          </Card>
        </div>
      </Phase>

      <ConfirmDialog open={confirm === 'stage'} onClose={() => setConfirm(null)} onConfirm={commit} busy={busy} error={err}
        title={stage ? `تأكيد ${stageMeta[stage].label} ${t.no}` : ''} confirmLabel="تأكيد"
        body={stage === 'ship' ? 'سيتم خصم الكميات المشحونة من رصيد المصدر وإضافتها كـ«قيد النقل» حتى يتم الاستلام.' : stage === 'receive' ? 'سيتم إضافة الكميات المستلمة إلى رصيد الوجهة وإغلاق التحويل.' : 'بعد الاعتماد لا يمكن تعديل بنود التحويل.'} />
      <ConfirmDialog open={confirm === 'cancel'} onClose={() => setConfirm(null)} onConfirm={cancel} busy={busy} danger title="إلغاء التحويل؟" confirmLabel="إلغاء التحويل"
        body={<>سيتم إلغاء <span className="mono c-primary">{t.no}</span> نهائيًا ولن يمكن إعادة تفعيله.</>} />
    </section>
  );
}

// Simulated fetch lifecycle: loading → (live | empty | error) per demo tweak
function useLoad(dep) {
  const { demoState, reloadKey } = useInv();
  const [phase, setPhase] = useState('loading');
  useEffect(() => {
    setPhase('loading');
    if (demoState === 'loading') return;
    const t = setTimeout(() => setPhase(demoState === 'live' ? 'ready' : demoState), 550);
    return () => clearTimeout(t);
  }, [demoState, reloadKey, dep]);
  return phase;
}

// Wrap list content with phase handling
function Phase({ phase, cols, rows, empty, children }) {
  if (phase === 'loading') return <SkeletonRows cols={cols} rows={rows} />;
  if (phase === 'error') return <ErrorState />;
  if (phase === 'empty') return empty;
  return children;
}

function InlineAlert({ kind = 'red', icon = 'alert', title, children, onClose }) {
  const map = { red: ['--badge-red-bg', '--badge-red-text'], amber: ['--badge-amber-bg', '--badge-amber-text'], blue: ['--badge-blue-bg', '--badge-blue-text'], green: ['--badge-green-bg', '--badge-green-text'] }[kind];
  return (
    <div className="rounded-2xl px-4 py-3 flex items-start gap-3 fade-in" style={{ background: `var(${map[0]})` }}>
      <span style={{ color: `var(${map[1]})` }} className="mt-0.5"><Icon n={icon} s={16} /></span>
      <div className="flex-1 text-[12.5px]">
        {title && <div className="font-semibold" style={{ color: `var(${map[1]})` }}>{title}</div>}
        <div className="c-primary opacity-80 mt-0.5">{children}</div>
      </div>
      {onClose && <button onClick={onClose} className="c-secondary"><Icon n="x" s={14} /></button>}
    </div>
  );
}

// ---------- Modal ----------
function Modal({ open, onClose, title, sub, children, footer, width = 'max-w-lg' }) {
  useEffect(() => {
    if (!open) return;
    const h = e => e.key === 'Escape' && onClose && onClose();
    window.addEventListener('keydown', h); return () => window.removeEventListener('keydown', h);
  }, [open]);
  if (!open) return null;
  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4 fade-in" style={{ background: 'rgba(0,0,0,.4)', backdropFilter: 'blur(4px)' }} onMouseDown={e => e.target === e.currentTarget && onClose && onClose()}>
      <div className={`v-card ${width} w-full shadow-2xl modal-in flex flex-col max-h-[92vh]`}>
        <div className="flex items-start justify-between gap-4 p-6 pb-4">
          <div>
            <h3 className="font-display text-[18px] font-semibold c-primary">{title}</h3>
            {sub && <p className="text-[12px] c-secondary mt-0.5">{sub}</p>}
          </div>
          <button onClick={onClose} className="w-8 h-8 rounded-full v-subcard flex items-center justify-center c-secondary shrink-0"><Icon n="x" s={14} /></button>
        </div>
        <div className="px-6 pb-2 overflow-y-auto flex-1">{children}</div>
        {footer && <div className="p-6 pt-4 flex items-center justify-start gap-2.5 flex-row-reverse">{footer}</div>}
      </div>
    </div>
  );
}

function ConfirmDialog({ open, onClose, onConfirm, title, body, confirmLabel = 'تأكيد', danger, busy, error }) {
  return (
    <Modal open={open} onClose={onClose} title={title} width="max-w-md"
      footer={<>
        <button className={`btn ${danger ? 'btn-orange' : 'btn-dark'}`} onClick={onConfirm} disabled={busy}>
          {busy ? <span className="w-3.5 h-3.5 rounded-full border-2 border-current border-t-transparent animate-spin" /> : null}
          {confirmLabel}
        </button>
        <button className="btn btn-soft" onClick={onClose}>تراجع</button>
      </>}>
      <div className="text-[13px] c-secondary leading-relaxed" style={{ textWrap: 'pretty' }}>{body}</div>
      {error && <div className="mt-4"><InlineAlert kind="red" title="رفض الخادم العملية">{error}</InlineAlert></div>}
    </Modal>
  );
}

// ---------- Variant picker: product → variant/SKU (never stops at product) ----------
function VariantPicker({ value, onChange, placeholder = 'ابحث بالاسم أو SKU أو الباركود', small, exclude = [], allowClear }) {
  const [open, setOpen] = useState(false);
  const [q, setQ] = useState('');
  const ref = useRef();
  useEffect(() => {
    const h = e => ref.current && !ref.current.contains(e.target) && setOpen(false);
    document.addEventListener('mousedown', h); return () => document.removeEventListener('mousedown', h);
  }, []);
  const v = value ? invVariant(value) : null;
  const ql = q.trim().toLowerCase();
  const groups = INV_PRODUCTS.map(p => ({
    ...p,
    variants: p.variants.filter(x => !exclude.includes(x.id) && (!ql || p.name.includes(q) || x.sku.toLowerCase().includes(ql) || x.barcode.includes(ql) || x.attrs.includes(q))),
  })).filter(p => p.variants.length);

  return (
    <div ref={ref} className="relative">
      <button type="button" onClick={() => setOpen(o => !o)} className={`inp ${small ? 'inp-sm' : ''} flex items-center justify-between gap-2 text-right`}>
        {v ? (
          <span className="flex items-center gap-2 min-w-0">
            <span className="truncate c-primary">{v.productName}</span>
            <span className="text-[11px] c-muted shrink-0">{v.attrs}</span>
            <span className="mono text-[10.5px] px-1.5 py-0.5 rounded-md shrink-0" style={{ background: 'var(--icon-well-bg)' }}>{v.sku}</span>
          </span>
        ) : <span className="c-muted truncate">{placeholder}</span>}
        <span className="flex items-center gap-1 shrink-0">
          {allowClear && v && <span onClick={e => { e.stopPropagation(); onChange(null); }} className="c-muted hover:c-primary"><Icon n="x" s={12} /></span>}
          <Icon n="chevD" s={13} className="c-muted" />
        </span>
      </button>
      {open && (
        <div className="absolute z-40 mt-1.5 w-full min-w-[320px] v-card shadow-2xl p-2 modal-in" style={{ borderRadius: 16 }}>
          <div className="v-subcard px-3 py-2 rounded-xl flex items-center gap-2 mb-1.5">
            <Icon n="search" s={14} className="c-muted" />
            <input autoFocus value={q} onChange={e => setQ(e.target.value)} placeholder="اسم المنتج، SKU، باركود…" className="bg-transparent text-[12.5px] outline-none flex-1 c-primary" />
            <Icon n="barcode" s={14} className="c-muted" />
          </div>
          <div className="max-h-[260px] overflow-y-auto">
            {groups.length === 0 && <div className="text-center text-[12px] c-muted py-6">لا توجد نتائج مطابقة</div>}
            {groups.map(p => (
              <div key={p.id} className="mb-1">
                <div className="text-[11px] font-semibold c-secondary px-2.5 pt-2 pb-1 flex items-center gap-1.5"><Icon n="box" s={12} /> {p.name} <span className="c-muted font-normal">· {p.variants.length} متغير</span></div>
                {p.variants.map(x => (
                  <button type="button" key={x.id} onClick={() => { onChange(x.id); setOpen(false); setQ(''); }}
                    className={`w-full flex items-center justify-between gap-3 px-2.5 py-2 rounded-xl text-right hover:bg-[var(--bg-subcard)] ${value === x.id ? 'bg-[var(--bg-subcard)]' : ''}`}>
                    <span className="text-[12.5px] c-primary pr-4">{x.attrs}</span>
                    <span className="flex items-center gap-2">
                      <span className="mono text-[10.5px] c-muted">{x.barcode}</span>
                      <span className="mono text-[11px] font-semibold c-primary">{x.sku}</span>
                      {value === x.id && <Icon n="check" s={13} className="c-orange" />}
                    </span>
                  </button>
                ))}
              </div>
            ))}
          </div>
        </div>
      )}
    </div>
  );
}

function NumInput({ value, onChange, decimals = 4, integer, small, placeholder = '0', className = '', err }) {
  return (
    <input inputMode="decimal" value={value ?? ''} placeholder={placeholder}
      onChange={e => {
        let s = e.target.value.replace(/[^\d.]/g, '');
        if (integer) s = s.replace(/\./g, '');
        const parts = s.split('.');
        if (parts.length > 2) s = parts[0] + '.' + parts.slice(1).join('');
        if (parts[1] && parts[1].length > decimals) s = parts[0] + '.' + parts[1].slice(0, decimals);
        onChange(s);
      }}
      className={`inp ${small ? 'inp-sm' : ''} num text-left ${err ? 'err' : ''} ${className}`} />
  );
}
