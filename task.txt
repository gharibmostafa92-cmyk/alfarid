export interface PageMeta {

  per_page: number;
  total: number;
  from: number | null;

}
export interface Paginated<T> {

  meta: PageMeta;
}

/** يحوّل أي استجابة قائمة (paginated أو مصفوفة) إلى Paginated موحّد */
export function asPage<T>(r: AxiosResponse<any>): Paginated<T> {
  const body = r.data;
  if (body && Array.isArray(body.data) && body.meta) return body as Paginated<T>;
  const arr: T[] = Array.isArray(body) ? body : Array.isArray(body?.data) ? body.data : [];
  return { data: arr, meta: { current_page: 1, last_page: 1, per_page: arr.length, total: arr.length, from: arr.length ? 1 : null, to: arr.length } };
}

/** يفك { data: T } */
export function one<T>(r: AxiosResponse<any>): T {
  return (r.data && 'data' in r.data ? r.data.data : r.data) as T;
}

/** يحذف القيم الفارغة، ويحوّل boolean إلى 1/0 لأن Laravel `boolean` يرفض "true" */
export function cleanParams(params: object = {}): Record<string, string | number> {
  const out: Record<string, string | number> = {};
  for (const [k, v] of Object.entries(params)) {
    if (v === undefined || v === null || v === '') continue;
    out[k] = typeof v === 'boolean' ? (v ? 1 : 0) : (v as any);
  }
  return out;
}

/** FormData لرفع الملفات. spoofPut=true عند التعديل لأن PHP لا يقرأ multipart مع PUT */
export function toFormData(obj: Record<string, unknown>, spoofPut = false): FormData {
  const fd = new FormData();
  if (spoofPut) fd.append('_method', 'PUT');
  const put = (key: string, v: unknown) => {
    if (v === undefined) return;
    if (v === null) return fd.append(key, '');
    if (typeof v === 'boolean') return fd.append(key, v ? '1' : '0');
    if (v instanceof Blob) return fd.append(key, v);
    if (Array.isArray(v)) {
      v.forEach((item, i) => {
        if (item !== null && typeof item === 'object' && !(item instanceof Blob)) {
          Object.entries(item).forEach(([ik, iv]) => put(`${key}[${i}][${ik}]`, iv));
        } else put(`${key}[${i}]`, item);
      });
      return;
    }
    fd.append(key, String(v));
  };
  Object.entries(obj).forEach(([k, v]) => put(k, v));
  return fd;
}

/** CRUD عام لموارد apiResource */
export function crud<T, P extends object, Q extends object = Record<string, unknown>>(path: string) {
  return {
    list: (params?: Q, signal?: AbortSignal) =>
      api.get(path, { params: cleanParams(params), signal }).then((r) => asPage<T>(r)),
    show: (id: Id, signal?: AbortSignal) => api.get(`${path}/${id}`, { signal }).then((r) => one<T>(r)),
    create: (body: P) => api.post(path, body).then((r) => one<T>(r)),
    update: (id: Id, body: Partial<P>) => api.put(`${path}/${id}`, body).then((r) => one<T>(r)),
    remove: (id: Id) => api.delete<{ message: string }>(`${path}/${id}`).then((r) => r.data),
  };
}

/** ===== معالجة الأخطاء ===== */
export interface ParsedApiError {
  status: number | null;
  message: string;
  fieldErrors: Record<string, string>; // أول رسالة لكل حقل
}
export function parseApiError(e: unknown): ParsedApiError {
  const err = e as any;
  const status: number | null = err?.response?.status ?? null;
  const body = err?.response?.data ?? {};
  const fieldErrors: Record<string, string> = {};
  if (body.errors && typeof body.errors === 'object') {
    for (const [k, v] of Object.entries(body.errors)) fieldErrors[k] = Array.isArray(v) ? String(v[0]) : String(v);
  }
  let message = body.message as string | undefined;
  if (!message) {
    if (status === null) message = 'تعذّر الاتصال بالخادم';
    else if (status === 403) message = 'ليست لديك صلاحية لتنفيذ هذا الإجراء';
    else if (status === 404) message = 'السجل غير موجود';
    else if (status >= 500) message = 'حدث خطأ في الخادم، حاول لاحقاً';
    else message = 'تعذّر تنفيذ الطلب';
  }
  return { status, message, fieldErrors };
}

Copyimport { useCallback, useEffect, useRef, useState, type DependencyList } from 'react';
import { isAxiosCancel, parseApiError, type ParsedApiError } from '@alfarid/api-client';
import { USE_MOCK, FALLBACK_ON_ERROR } from './source';

export interface RemoteState<T> {
  data: T | null;
  loading: boolean;
  error: ParsedApiError | null;
  usingMock: boolean;
  reload: () => void;
  setData: React.Dispatch<React.SetStateAction<T | null>>;
}

export function useRemote<T>(
  fetcher: (signal: AbortSignal) => Promise<T>,
  deps: DependencyList,
  opts: { enabled?: boolean; mock?: () => T; onError?: (e: ParsedApiError) => void } = {},
): RemoteState<T> {
  const { enabled = true, mock, onError } = opts;
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState<boolean>(enabled);
  const [error, setError] = useState<ParsedApiError | null>(null);
  const [usingMock, setUsingMock] = useState(false);
  const [tick, setTick] = useState(0);
  const onErrorRef = useRef(onError);
  onErrorRef.current = onError;

  useEffect(() => {
    if (!enabled) { setLoading(false); return; }
    if (USE_MOCK && mock) { setData(mock()); setUsingMock(true); setLoading(false); return; }
    const ctrl = new AbortController();
    setLoading(true);
    setError(null);
    fetcher(ctrl.signal)
      .then((d) => { setData(d); setUsingMock(false); })
      .catch((e) => {
        if (isAxiosCancel(e)) return;
        const pe = parseApiError(e);
        setError(pe);
        onErrorRef.current?.(pe);
        if (FALLBACK_ON_ERROR && mock && (pe.status === null || pe.status >= 500)) {
          setData(mock()); setUsingMock(true);
        }
      })
      .finally(() => { if (!ctrl.signal.aborted) setLoading(false); });
    return () => ctrl.abort();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [...deps, tick, enabled]);

  const reload = useCallback(() => setTick((t) => t + 1), []);
  return { data, loading, error, usingMock, reload, setData };
}

/** تأخير قيمة البحث 350ms */
export function useDebounced<T>(value: T, ms = 350): T {
  const [v, setV] = useState(value);
  useEffect(() => { const t = setTimeout(() => setV(value), ms); return () => clearTimeout(t); }, [value, ms]);
  return v;
}

/** تنفيذ mutation مع حالة busy وtoast */
export function useAction() {
  const [busy, setBusy] = useState(false);
  const run = useCallback(async <R,>(
    fn: () => Promise<R>,
    h: { toast: (t: string, tone?: 'accent' | 'green' | 'red') => void; success?: string; onFieldErrors?: (f: Record<string, string>) => void },
  ): Promise<R | undefined> => {
    setBusy(true);
    try {
      const r = await fn();
      if (h.success) h.toast(h.success, 'green');
      return r;
    } catch (e) {
      const pe = parseApiError(e);
      if (pe.status === 422 && Object.keys(pe.fieldErrors).length && h.onFieldErrors) h.onFieldErrors(pe.fieldErrors);
      h.toast(pe.message, 'red');
      return undefined;
    } finally {
      setBusy(false);
    }
  }, []);
  return { busy, run };
}

export function useFormState<T extends Record<string, any>>(initial: T) {
  const [v, setV] = useState<T>(initial);
  const [errors, setErrors] = useState<Record<string, string>>({});
  const set = <K extends keyof T>(k: K, val: T[K]) => {

    set(k, val as T[keyof T]);
  };
  return { v, set, on, errors, setErrors };

شركات الشحن	/shipping-companies (apiResource)	api_credentials يُدمج ولا يُستبدل؛ وإرسال null يتجاهله الخادم
مناطق الشحن	/shipping-zones	المدن تُرسل كـ city_ids: number[]
أسعار الشحن	/shipping-rates	الاستجابة تحمّل company وzone
مجموعات العملاء	/customer-groups	فلاتر search, is_active, per_page(≤100)؛ الإنشاء والتعديل للمستخدم المركزي فقط (403 لغيره)
قوائم الأسعار	/price-lists	فلاتر type, customer_id, customer_group_id, branch_id, is_active, search, per_page
استيراد CSV	POST /price-lists/{id}/import	الحقل اسمه file، والرد {message, data: report}
تصدير CSV	GET /price-lists/{id}/export	StreamedResponse
بنود القائمة	/price-lists/{id}/items (nested apiResource)	scopeBindings
العروض	/promotions	value/applies_to/is_active بقيمة null تُحذف من الـ payload في الخادم
الكوبونات	/coupons	لا يمكن حذف كوبون له استخدامات (BusinessException)
البراندات	/brands	استخدم brandsApi الموجود (multipart + _method=PUT)
الصلاحيات المطلوبة: shipping.*، customer_groups.*، brands.*. وقد يعيد الخادم 403، فيُعرض عبر toast.

Copy// mappers.ts
export async function optimisticToggle<T extends { id: number }>(
  list: T[], setList: (l: T[]) => void, id: number, patch: Partial<T>, call: () => Promise<unknown>,
  toast: (t: string, tone?: 'accent' | 'green' | 'red') => void,
) {
  const prev = list;
  setList(list.map((x) => (x.id === id ? { ...x, ...patch } : x)));
  try { await call(); } catch (e) { setList(prev); toast(parseApiError(e).message, 'red'); }
}

export const MAX_REPORT_DAYS = 366;
export function validatePeriod(from?: string, to?: string): string | null {
  if (!from || !to) return null; // الخادم يستخدم بداية الشهر الحالي → اليوم
  const a = new Date(from + 'T00:00:00Z').getTime(), b = new Date(to + 'T00:00:00Z').getTime();
  if (Number.isNaN(a) || Number.isNaN(b)) return 'صيغة التاريخ غير صحيحة';
  if (b < a) return 'تاريخ النهاية يجب أن يكون بعد تاريخ البداية';
  if ((b - a) / 86_400_000 > MAX_REPORT_DAYS) return `أقصى مدى للتقرير ${MAX_REPORT_DAYS} يوماً`;
  return null;
}

Copyimport { useCallback, useEffect, useRef, useState } from 'react';
import { notificationsApi, type NotificationApi, parseApiError, isAxiosCancel } from '@alfarid/api-client';
import type { NotificationItem } from '../data';
import { USE_MOCK } from './source';
import { NOTIFICATIONS } from '../data'; // VERIFY اسم المصفوفة الوهمية

const POLL_MS = 60_000;
const toItem = (n: NotificationApi): NotificationItem => ({ id: n.id, type: n.type, data: n.data, read_at: n.read_at, created_at: n.created_at ?? '' });

export function useNotifications(onError?: (msg: string) => void) {
  const [items, setItems] = useState<NotificationItem[]>(USE_MOCK ? NOTIFICATIONS : []);
  const [unread, setUnread] = useState<number>(USE_MOCK ? NOTIFICATIONS.filter((n) => !n.read_at).length : 0);
  const [page, setPage] = useState(1);
  const [lastPage, setLastPage] = useState(1);
  const [loading, setLoading] = useState(!USE_MOCK);
  const ctrlRef = useRef<AbortController | null>(null);

  const load = useCallback(async (p = 1) => {
    if (USE_MOCK) return;
    ctrlRef.current?.abort();
    const ctrl = new AbortController(); ctrlRef.current = ctrl;
    try {
      const r = await notificationsApi.list(p, ctrl.signal);
      setItems((prev) => (p === 1 ? r.data.map(toItem) : [...prev, ...r.data.map(toItem)]));
      setUnread(r.meta.unread_count); setPage(r.meta.current_page); setLastPage(r.meta.last_page);
    } catch (e) {
      if (!isAxiosCancel(e)) onError?.(parseApiError(e).message);
    } finally { setLoading(false); }
  }, [onError]);

  // تحميل أولي + polling فقط عندما يكون التبويب ظاهراً
  useEffect(() => {
    load(1);
    const t = setInterval(() => { if (document.visibilityState === 'visible') load(1); }, POLL_MS);
    const onVis = () => { if (document.visibilityState === 'visible') load(1); };
    document.addEventListener('visibilitychange', onVis);
    return () => { clearInterval(t); document.removeEventListener('visibilitychange', onVis); ctrlRef.current?.abort(); };
  }, [load]);

  const markRead = useCallback(async (id: string) => {
    const now = new Date().toISOString();
    setItems((l) => l.map((n) => (n.id === id && !n.read_at ? { ...n, read_at: now } : n)));      // optimistic
    if (USE_MOCK) { setUnread((u) => Math.max(0, u - 1)); return; }
    try { const r = await notificationsApi.markRead(id); setUnread(r.meta.unread_count); }
    catch (e) { setItems((l) => l.map((n) => (n.id === id ? { ...n, read_at: null } : n))); onError?.(parseApiError(e).message); }
  }, [onError]);

  /** لا يوجد endpoint "قراءة الكل" — نمرّ على غير المقروء في المحمّل حالياً بالتتابع */
  const markAllRead = useCallback(async () => {
    for (const n of items.filter((x) => !x.read_at)) await markRead(n.id);
  }, [items, markRead]);

  const loadMore = useCallback(() => { if (page < lastPage) load(page + 1); }, [page, lastPage, load]);

  return { items, unread, loading, markRead, markAllRead, loadMore, hasMore: page < lastPage, reload: () => load(1) };
}
