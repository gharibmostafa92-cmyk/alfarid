Copyconst MULTIPART = { headers: { 'Content-Type': 'multipart/form-data' } };

function categoryFormData(p: CategoryPayload, spoofPut = false): FormData {
  const fd = new FormData();
  if (spoofPut) fd.append('_method', 'PUT');
  const put = (key: string, v: unknown) => {
    if (v === undefined) return;                       // مش هيتبعت
    if (v === null) return fd.append(key, '');         // Laravel يحوّلها null
    if (typeof v === 'boolean') return fd.append(key, v ? '1' : '0'); // مهم: مش "true"
    if (v instanceof Blob) return fd.append(key, v);
    fd.append(key, String(v));
  };
  put('parent_id', p.parent_id);
  put('name_ar', p.name_ar);
  put('name_en', p.name_en);
  put('slug', p.slug);
  put('description_ar', p.description_ar);
  put('description_en', p.description_en);
  put('sort_order', p.sort_order);
  put('is_active', p.is_active);
  if (p.image) put('image', p.image);
  if (p.remove_image) put('remove_image', true);
  return fd;
}

// تحويل الشجرة لـ items الخاصة بالـ reorder — بنبعت اللي اتغير بس
export function buildReorderItems(tree, savedTree) {
  const flat = (nodes, out = new Map(), parentId = null) => {
    nodes.forEach((n, i) => {
      out.set(n.id, { id: n.id, parent_id: parentId, sort_order: i });
      flat(n.children || [], out, n.id);
    });
    return out;
  };
  const now = flat(tree);
  const before = flat(savedTree);
  return [...now.values()].filter((it) => {
    const b = before.get(it.id);
    return !b || b.parent_id !== it.parent_id || b.sort_order !== it.sort_order;
  });
}

export const toLatinDigits = (s) =>
  String(s ?? '')
    .replace(/[٠-٩]/g, (d) => '٠١٢٣٤٥٦٧٨٩'.indexOf(d))
    .replace(/[۰-۹]/g, (d) => '۰۱۲۳۴۵۶۷۸۹'.indexOf(d));

// تطبيع للبحث: يشيل التشكيل والتطويل ويوحّد الألف والياء والتاء المربوطة
export const normalizeSearch = (s) =>
  String(s ?? '')
    .toLowerCase()
    .replace(/[\u064B-\u0652\u0640]/g, '')
    .replace(/[أإآ]/g, 'ا')
    .replace(/ى/g, 'ي')
    .replace(/ة/g, 'ه')
    .trim();

// الاسم المعروض مع fallback لو لغة فاضية
export const displayName = (n, lang = 'ar') =>
  (lang === 'ar' ? n.ar || n.en : n.en || n.ar) || '—';

// slug تلقائي من الإنجليزي فقط (لو الناتج فاضي → نسيبه null والباك يولّد)
export const autoSlug = (en) =>
  String(en ?? '').toLowerCase().trim().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');

export function parseApiError(err) {
  const status = err?.response?.status ?? 0;
  const data = err?.response?.data ?? {};
  const fieldErrors = {};
  Object.entries(data.errors ?? {}).forEach(([k, msgs]) => {
    const key = FIELD_MAP[k] ?? (k.startsWith('items') ? 'items' : k);
    fieldErrors[key] = Array.isArray(msgs) ? msgs[0] : String(msgs);
  });
  return {
    status,
    message: data.message || FALLBACK[status] || FALLBACK[500],
    fieldErrors,
    isValidation: status === 422 && Object.keys(fieldErrors).length > 0,
  };
}
