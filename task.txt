Copy<?php

declare(strict_types=1);

namespace App\Support;

use Illuminate\Support\Facades\Storage;

final class BannerImage
{
    public static function disk(): string
    {
        return (string) config('filesystems.banners_disk', 'r2');
    }

    /** يرجع null بدل ما يكسر الـ endpoint لو التخزين فيه مشكلة */
    public static function url(?string $path): ?string
    {
        if ($path === null || $path === '') {
            return null;
        }
        try {
            $disk = Storage::disk(self::disk());
            if ($disk->providesTemporaryUrls()) {
                $ttl = max(60, min(604800, (int) config('filesystems.disks.r2.download_url_ttl_seconds', 3600)));
                return $disk->temporaryUrl($path, now()->addSeconds($ttl));
            }
            return $disk->url($path);
        } catch (\Throwable $e) {
            report($e);
            return null;
        }
    }
}

1.4 تحويل link_value لرابط جاهز (href)
الملف: app/Services/BannerLinkResolver.php. الفكرة إننا نحمّل كل المنتجات والأقسام المطلوبة في query واحدة، من غير N+1.

Copy<?php

declare(strict_types=1);

namespace App\Services;

use App\Enums\BannerLinkType;
use App\Models\Category;
use Illuminate\Support\Collection;

final class BannerLinkResolver
{
    public function __construct(private readonly StoreCatalogService $catalog) {}

    /** @param Collection<int, \App\Models\Banner> $banners */
    public function attach(Collection $banners): Collection
    {
        $productIds = $banners->filter(fn ($b) => $b->link_type === BannerLinkType::Product)
            ->pluck('link_value')->filter()->map(fn ($v) => (int) $v)->unique()->values()->all();
        $categoryIds = $banners->filter(fn ($b) => $b->link_type === BannerLinkType::Category)
            ->pluck('link_value')->filter()->map(fn ($v) => (int) $v)->unique()->values()->all();

        // منتج مش منشور = البانر يظهر بس من غير رابط (مش هنودّي العميل لصفحة 404)
        $productSlugs = $productIds === [] ? collect() : $this->catalog->publishedQuery()
            ->whereIn('products.id', $productIds)->pluck('products.slug', 'products.id');
        $categorySlugs = $categoryIds === [] ? collect() : Category::query()
            ->whereIn('id', $categoryIds)->where('is_active', true)->pluck('slug', 'id');

        return $banners->each(function ($b) use ($productSlugs, $categorySlugs): void {
            $href = match ($b->link_type) {
                BannerLinkType::Product => ($s = $productSlugs->get((int) $b->link_value)) ? "/products/{$s}" : null,
                BannerLinkType::Category => ($s = $categorySlugs->get((int) $b->link_value)) ? "/shop?category={$s}" : null,
                BannerLinkType::External => $b->link_value ?: null,
                default => null, // None و Brand (Brand ميزة قديمة والمتجر مش بيفلتر بيها)
            };
            $b->setAttribute('resolved_href', $href);
        });
    }
}

Copyfinal class BannerService
{
    public function create(array $data, UploadedFile $image): Banner
    {
        $path = $this->storeImage($image);
        try {
            return DB::transaction(fn () => Banner::create([...$this->normalize($data), 'image_path' => $path]));
        } catch (\Throwable $e) {
            Storage::disk(BannerImage::disk())->delete($path); // ميبقاش فيه ملفات يتيمة
            throw $e;
        }
    }

    public function update(Banner $banner, array $data, ?UploadedFile $image): Banner
    {
        $old = null;
        if ($image) {
            $data['image_path'] = $this->storeImage($image);
            $old = $banner->image_path;
        }
        DB::transaction(fn () => $banner->update($this->normalize($data)));
        if ($old) {
            Storage::disk(BannerImage::disk())->delete($old);
        }
        return $banner->refresh();
    }

    public function delete(Banner $banner): void
    {
        $path = $banner->image_path;
        $banner->delete();
        Storage::disk(BannerImage::disk())->delete($path);
    }

    /** @param list<int> $ids */
    public function reorder(string $position, array $ids): void
    {
        DB::transaction(function () use ($position, $ids): void {
            foreach (array_values($ids) as $i => $id) {
                Banner::where('id', $id)->where('position', $position)->update(['sort_order' => $i + 1]);
            }
        });
        Cache::forget('store_home_data'); // update() الجماعي مبيشغّلش model events
        Cache::forget('illuminate:cache:flexible:created:store_home_data');
    }

    private function normalize(array $d): array
    {
        unset($d['image']);
        if (in_array($d['link_type'] ?? 'none', ['none'], true)) {
            $d['link_value'] = null;
        }
        if (isset($d['link_value'])) {
            $d['link_value'] = trim((string) $d['link_value']);
        }
        return $d;
    }

    private function storeImage(UploadedFile $file): string
    {
        return $file->storeAs('banners', Str::uuid().'.'.$file->extension(), BannerImage::disk());
    }
}

export const isoToLocalInput = (iso?: string | null) => {
  if (!iso) return '';

};
export const localInputToIso = (v?: string) => (v ? new Date(v).toISOString() : '');

export function fromApi(b: ApiBanner): BannerItem {
  return {
    id: String(b.id), title: b.title, placement: POS_FROM_API[b.position],
    image: b.image_url, linkType: b.link_type,

    url: b.link_type === 'external' ? b.link_value ?? '' : '',
    linkLabel: b.link_label, sort: b.sort_order,
    start: isoToLocalInput(b.starts_at), end: isoToLocalInput(b.ends_at),
    active: b.is_active, status: b.status,
    kicker: b.kicker ?? '', headline: b.headline ?? '', headlineBold: b.headline_bold ?? '',

}

export function toFormData(f: BannerItem & { imageFile?: File | null }): FormData {
  const fd = new FormData();
  fd.append('title', f.title.trim());
  if (f.imageFile) fd.append('image', f.imageFile);
  fd.append('position', POS_TO_API[f.placement]);
  fd.append('link_type', f.linkType);
  fd.append('link_value', f.linkType === 'external' ? f.url.trim() : f.linkType === 'none' ? '' : f.linkId ?? '');
  fd.append('sort_order', String(f.sort));
  fd.append('starts_at', localInputToIso(f.start));
  fd.append('ends_at', localInputToIso(f.end));
  fd.append('is_active', f.active ? '1' : '0');
  (['kicker', 'headline', 'description'] as const).forEach((k) => fd.append(k, (f as any)[k] ?? ''));
  fd.append('headline_bold', f.headlineBold ?? '');
  fd.append('cta_label', f.ctaLabel ?? '');
  fd.append('bg_color', f.bgColor ?? '');
  return fd;
}

export function cardToSophie(c: StoreProductCard, mode: 'new' | 'sale' | 'auto'): SophieProduct | null {
  const price = num(c.price);
  if (!Number.isFinite(price)) return null;            // منتج من غير سعر = ميتعرضش
  const old = num(c.old_price);
  let badge: string | undefined; let badgeTone: SophieProduct['badgeTone'];
  if (!c.in_stock) { badge = 'SOLD OUT'; badgeTone = 'muted'; }
  else if (c.discount_percentage) { badge = `-${Math.round(c.discount_percentage)}%`; badgeTone = 'sale'; }
  else if (mode === 'new') { badge = 'NEW'; badgeTone = 'default'; }
  return {
    id: c.id, slug: c.slug,
    name: c.name_en?.trim() || c.name_ar,
    price, oldPrice: Number.isFinite(old) && old > price ? old : undefined,
    badge, badgeTone,
    imgUrl: c.image || null,                            // مفيش tintClass للصور الحقيقية
    category: c.category?.name_en || c.category?.name_ar,
    href: STORE_ROUTES.product(c.slug), inStock: c.in_stock,
  };
}

export const mapCards = (list: StoreProductCard[] | undefined, mode: 'new' | 'sale' | 'auto') =>
  (list ?? []).map((c) => cardToSophie(c, mode)).filter(Boolean) as SophieProduct[];
